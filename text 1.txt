local function u47()

	local char = lp.Character or lp.CharacterAdded:Wait()
	local root = char:WaitForChild("HumanoidRootPart")
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hum then return end
	hum.AutoRotate = false
	t4.bodyGyro = Instance.new("BodyGyro")
	t4.bodyGyro.P = 90000
	t4.bodyGyro.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
	t4.bodyGyro.CFrame = root.CFrame
	t4.bodyGyro.Parent = root
	t4.bodyVel = Instance.new("BodyVelocity")
	t4.bodyVel.MaxForce = Vector3.new(9e9, 9e9, 9e9)
	t4.bodyVel.Velocity = Vector3.zero
	t4.bodyVel.Parent = root
	t4.flightConn = RunService.Heartbeat:Connect(function()
		local cam = workspace.CurrentCamera
		local lv, rv = cam.CFrame.LookVector, cam.CFrame.RightVector
		local md = hum.MoveDirection
		local fwd = md:Dot(Vector3.new(lv.X, 0, lv.Z).Unit)
		local right = md:Dot(Vector3.new(rv.X, 0, rv.Z).Unit)
		local vy = md.Magnitude > 0 and lv.Y * fwd or 0
		local v = lv * fwd + rv * right
		local dir = Vector3.new(v.X, vy, v.Z)
		if dir.Magnitude > 1 then dir = dir.Unit end
		t4.bodyVel.Velocity = dir * t3.flySpeed
		t4.bodyGyro.CFrame = CFrame.lookAt(root.Position, root.Position + lv, cam.CFrame.UpVector)
	end)
end

local function uAntiSlowCheck(v)
	if not t1.antiSlownessActive or not v:IsA("NumberValue") then return end
	if v.Name == "Sprinting" or v.Name:lower() == "emoting" then return end
	if v.Name == "DirectionalMovement" or v.Name == "FixingGenerator" or v.Name:upper() == "ENRAGED" then
		if v.Value < 1 then v.Value = 1 end
	elseif v.Value > 0.05 and v.Value < 1 then
		pcall(function() v:Destroy() end)
	end
end

local INV = { active = false, underground = false, originalNamecall = nil, unreliable = nil }
local function invInstallHook()
	if INV.originalNamecall then return true end
	if not capOK("hookmetamethod", "checkcaller", "newcclosure") then
		capWarn("Invincible", { "hookmetamethod", "checkcaller", "newcclosure" })
		return false
	end
	local net = game.ReplicatedStorage:FindFirstChild("Modules")
	net = net and net:FindFirstChild("Network", true)
	INV.unreliable = net and net:FindFirstChild("UnreliableRemoteEvent")
	if not INV.unreliable then capWarn("Invincible", { "UnreliableRemoteEvent" }); return false end
	local ok, res = pcall(function()
		local __nc
		__nc = CAP.hookmetamethod(game, "__namecall", CAP.newcclosure(function(self, ...)
			if not CAP.checkcaller() and INV.underground and getnamecallmethod() == "FireServer" and self == INV.unreliable then
				local args = { ... }
				if args[1] == 1 then return function() return nil end end
			end
			return __nc(self, ...)
		end))
		return __nc
	end)
	if ok and res then INV.originalNamecall = res; return true end
	return false
end
local function invIsClear(hitbox, pos)
	if not hitbox or not pos then return false end
	local p = OverlapParams.new()
	p.FilterType = Enum.RaycastFilterType.Include
	p.MaxParts = 1
	p.FilterDescendantsInstances = { hitbox }
	return #workspace:GetPartBoundsInRadius(pos, 2.5, p) == 0
end
local function invVelocity(target)
	local root = Players.LocalPlayer.Character and Players.LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
	if not root then return end
	local body = Instance.new("BodyVelocity")
	body.MaxForce = Vector3.new(9e9, 9e9, 9e9)
	body.Velocity = Vector3.zero
	body.Parent = root
	local t0 = workspace.DistributedGameTime + 7
	while (root.Position - target).Magnitude > 2 and workspace.DistributedGameTime < t0 do
		body.Velocity = (target - root.Position).Unit * 100
		RunService.RenderStepped:Wait()
	end
	body:Destroy()
end
local function invGoUnder(on)
	local char = Players.LocalPlayer.Character
	if not char then return end
	local root = char:FindFirstChild("HumanoidRootPart")
	local head = char:FindFirstChild("Head")
	local hum = char:FindFirstChildOfClass("Humanoid")
	local hitbox = char:FindFirstChild("QueryHitbox")
	if not (root and head and hum) then return end
	if on then
		if INV.underground then return true end
		local old = root.CFrame
		local under = old * CFrame.new(0, -22, 0)
		hum.CameraOffset = Vector3.new(0, 12e12, 0)
		task.wait(0.1)
		root.CFrame = under
		for _ = 1, 3 do
			root.Velocity = Vector3.zero
			invVelocity(under.Position)
			head.Anchored = true
			repeat task.wait() until invIsClear(hitbox, old.Position)
			INV.underground = true
			task.wait()
			head.Anchored = false
			root.CFrame = old
			root.Velocity = Vector3.zero
			if invIsClear(hitbox, old.Position) then break end
		end
		if not invIsClear(hitbox, old.Position) then
			INV.underground = false
			hum.CameraOffset = Vector3.zero
			return false
		end
		return true
	else
		INV.underground = false
		local hum = char:FindFirstChildOfClass("Humanoid")
		if hum then hum.CameraOffset = Vector3.zero end
		if head then head.Anchored = false end
		return true
	end
end

	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not hrp then return nil, math.huge end
	local s = workspace.Players:FindFirstChild("Survivors")
	if not s then return nil, math.huge end
	local best, huge = nil, math.huge
	for _, c in pairs(s:GetChildren()) do
		local r = c:FindFirstChild("HumanoidRootPart")
		if r then
			local m = (hrp.Position - r.Position).Magnitude
			if m < huge then huge = m; best = c end
		end
	end
	return best, huge
end

local PLUGIN_DIR = "outlawdev/plugins"
task.spawn(function()
	if not capOK("isfolder", "makefolder", "listfiles", "readfile", "setfenv") then
		capWarn("Plugin Loader", { "isfolder", "makefolder", "listfiles", "readfile", "setfenv" })
		return
	end
	pcall(function()
		if not CAP.isfolder("outlawdev") then CAP.makefolder("outlawdev") end
		if not CAP.isfolder(PLUGIN_DIR) then CAP.makefolder(PLUGIN_DIR) end
	end)
	local files = CAP.listfiles(PLUGIN_DIR)
	if not files then return end
	for _, path in ipairs(files) do
		local ok, data = pcall(CAP.readfile, path)
		if not ok or type(data) ~= "string" then goto continue end
		if data:gsub("[\n ]", ""):sub(1, 7) ~= "return{" then
			warn("[outlaw.dev] plugin rejected (must start with return{): " .. tostring(path))
			goto continue
		end
		local chunk = loadstring(data)
		if not chunk then goto continue end
		local ok2, features = pcall(function()
			CAP.setfenv(chunk, getfenv())
			return chunk()
		end)
		if not ok2 or type(features) ~= "table" then goto continue end
		for tabName, tabContents in pairs(features) do
			if type(tabName) == "string" and type(tabContents) == "table" then
				-- plugins append to existing tabs by name; new tabs skipped in this build
				local tabMap = {
					Automation = v26, Features = v23, Visuals = v24,
					Miscellaneous = v30, Killer = v27, Survivor = v26,
				}
				local tab = tabMap[tabName]
				if not tab then goto nextTab end
				for featName, featData in pairs(tabContents) do
					if featName == "TabAttributes" then goto nextFeat end
					if type(featData) ~= "table" then goto nextFeat end
					if featData.InstanceType == "BoolValue" then
						tab:Toggle({
							Title = tostring(featData.DisplayTitle or featName),
							Desc = tostring(featData.DisplayDescription or ""),
							Value = featData.DefaultInstanceValue == true,
							Callback = function(state)
								if type(featData.ScriptFunction) == "function" then
									pcall(featData.ScriptFunction, featData, state)
								end
							end,
						})
					elseif featData.InstanceType == "NumberValue" then
						local ed = featData.ExtraData or {}
						tab:Slider({
							Title = tostring(featData.DisplayTitle or featName),
							Desc = tostring(featData.DisplayDescription or ""),
							Value = { Min = ed.MinValue or 0, Max = ed.MaxValue or 100, Default = featData.DefaultInstanceValue or 0, Step = ed.Step or 1 },
							Callback = function(v)
								if type(featData.ScriptFunction) == "function" then
									pcall(featData.ScriptFunction, featData, v)
								end
							end,
						})
					end
					::nextFeat::
				end
			end
			::nextTab::
		end

task.spawn(function()
	local scriptVersion = "2026-10-04"
	local ok, res = pcall(function()
		return game:HttpGet("https://apis.rovalra.com/v1/games/history?place_id=18687417158")
	end)
	if not ok or not res then return end
	local dec = pcall(function() return HttpService:JSONDecode(tostring(res)) end)
	if not dec then return end
	local ok2, data = pcall(function() return HttpService:JSONDecode(tostring(res)) end)
	if not ok2 or type(data) ~= "table" then return end
	local hist = data.history and data.history[1]
	local gameVer = hist and hist.first_seen and hist.first_seen:sub(1, 10)
	if not gameVer then return end
	local y1, m1, d1 = scriptVersion:match("(%d+)-(%d+)-(%d+)")
	local y2, m2, d2 = gameVer:match("(%d+)-(%d+)-(%d+)")
	if not (y1 and y2) then return end
	local tScript = os.time({ year = y1, month = m1, day = d1 })
	local tGame = os.time({ year = y2, month = m2, day = d2 })
	local daysSinceScript = math.floor(math.abs(tScript - os.time()) / 86400)
	local daysSinceGame = math.floor(math.abs(tGame - os.time()) / 86400)
	if tScript < tGame and daysSinceScript > 1 then
		v30:Section({ Title = u16("SCRIPT OUTDATED", t10.Danger, t10.Text) })
		v30:Paragraph({
			Title = u16("Game updated " .. daysSinceGame .. "d ago. Script tested " .. daysSinceScript .. "d ago.", t10.Danger, t10.Text),
			Desc = "Some features may not work. Wait for a script update.",
		})
		u1:Notify({ Title = "outlaw.dev", Content = "Script is older than game build", Duration = 8 })
	end
end)
